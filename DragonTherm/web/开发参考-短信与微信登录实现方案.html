2.3 SmsCodeService — 改造后完整代码 核心

替换原有的 sendCode() 方法。与旧代码相比，仅将 System.out.println 替换为真实的阿里云短信 API 调用，其余校验逻辑完全不变。

Javasrc/main/java/cn/puge/service/SmsCodeService.java
package cn.puge.service;

import cn.puge.config.SmsConfig;
import cn.puge.exception.BusinessException;

import com.aliyuncs.DefaultAcsClient;
import com.aliyuncs.IAcsClient;
import com.aliyuncs.dysmsapi.model.v20170525.SendSmsRequest;
import com.aliyuncs.dysmsapi.model.v20170525.SendSmsResponse;
import com.aliyuncs.profile.DefaultProfile;
import com.aliyuncs.profile.IClientProfile;

import java.util.Map;
import java.util.Random;
import java.util.concurrent.ConcurrentHashMap;

/**
 * 短信验证码服务（生产版）
 * 接入阿里云短信网关，实现真实短信发送
 */
public class SmsCodeService {

    private static final Map<String, SmsCodeInfo> SMS_CODE_MAP =
        new ConcurrentHashMap<>();

    // ========== 验证码信息内部类（保持不变） ==========
    private static class SmsCodeInfo {
        private String code;
        private long sendTime;
        private boolean used;

        public SmsCodeInfo(String code, long sendTime) {
            this.code = code;
            this.sendTime = sendTime;
            this.used = false;
        }
        public String getCode()  { return code; }
        public long   getSendTime() { return sendTime; }
        public boolean isUsed()    { return used; }
        public void setUsed(boolean used) { this.used = used; }
    }

    // ========== 生成 6 位验证码（保持不变） ==========
    public String generateCode() {
        int code = new Random().nextInt(1_000_000);
        return String.format("%06d", code);
    }

    // ================================================================
    // ★★★ 核心改造方法：sendCode() — 对接阿里云短信 API ★★★
    // ================================================================
    public String sendCode(String phone) {
        // ① 参数校验（保持不变）
        if (phone == null || phone.trim().isEmpty()) {
            throw new BusinessException("PHONE_NULL", "手机号不能为空");
        }

        // ② 频率限制检查 — 同一手机号 60s 内不可重复发送（保持不变）
        SmsCodeInfo existing = SMS_CODE_MAP.get(phone);
        if (existing != null) {
            long elapsed = System.currentTimeMillis() - existing.getSendTime();
            if (elapsed < SmsConfig.RESEND_INTERVAL) {
                long remainSec = (SmsConfig.RESEND_INTERVAL - elapsed) / 1000;
                throw new BusinessException("SEND_TOO_FAST",
                    "验证码发送过于频繁，请" + remainSec + "秒后再试");
            }
        }

        // ③ 生成新验证码（保持不变）
        String code = generateCode();

        // ④ ★★★ 调用真实短信网关发送 ★★★
        boolean sendOk = sendSmsViaAliyun(phone, code);
        if (!sendOk) {
            throw new BusinessException("SMS_SEND_FAIL",
                "短信发送失败，请稍后重试");
        }

        // ⑤ 存入缓存（保持不变，不在日志中打印验证码明文）
        SMS_CODE_MAP.put(phone, new SmsCodeInfo(code, System.currentTimeMillis()));
        System.out.println("【短信】已向 " + maskPhone(phone) + " 发送验证码");

        return code;  // 仅内部使用，HttpServer 不应再暴露给前端
    }

    // ================================================================
    // ★★★ 新增方法：阿里云短信 API 调用 ★★★
    // ================================================================
    private boolean sendSmsViaAliyun(String phone, String code) {
        try {
            // 创建 Profile 和 Client
            IClientProfile profile = DefaultProfile.getProfile(
                SmsConfig.REGION_ID,
                SmsConfig.ACCESS_KEY_ID,
                SmsConfig.ACCESS_KEY_SECRET
            );
            IAcsClient client = new DefaultAcsClient(profile);

            // 构造请求
            SendSmsRequest request = new SendSmsRequest();
            request.setPhoneNumbers(phone);                   // 目标手机号
            request.setSignName(SmsConfig.SIGN_NAME);         // 短信签名
            request.setTemplateCode(SmsConfig.TEMPLATE_CODE); // 短信模板CODE

            // 模板变量 — JSON 格式，key 必须与短信模板中的 ${code} 一致
            request.setTemplateParam("{\"code\":\"" + code + "\"}");

            // 发送请求
            SendSmsResponse response = client.getAcsResponse(request);

            // 判断结果
            if ("OK".equals(response.getCode())) {
                return true;
            } else {
                System.err.println("短信发送失败: " + response.getMessage());
                return false;
            }
        } catch (Exception e) {
            System.err.println("短信发送异常: " + e.getMessage());
            return false;
        }
    }

    // ========== 验证码校验（完全不变） ==========
    public boolean verifyCode(String phone, String code) {
        if (phone == null || phone.trim().isEmpty())
            throw new BusinessException("PHONE_NULL", "手机号不能为空");
        if (code == null || code.trim().isEmpty())
            throw new BusinessException("CODE_NULL", "验证码不能为空");

        SmsCodeInfo info = SMS_CODE_MAP.get(phone);
        if (info == null)
            throw new BusinessException("CODE_NOT_EXIST", "验证码已过期或未发送");
        if (info.isUsed())
            throw new BusinessException("CODE_USED", "验证码已被使用");

        long elapsed = System.currentTimeMillis() - info.getSendTime();
        if (elapsed > SmsConfig.CODE_EXPIRE_MS) {
            SMS_CODE_MAP.remove(phone);
            throw new BusinessException("CODE_EXPIRED", "验证码已过期");
        }
        if (!info.getCode().equals(code))
            throw new BusinessException("CODE_MISMATCH", "验证码错误");

        SMS_CODE_MAP.remove(phone);
        return true;
    }

    // ========== 辅助方法 ==========
    private String maskPhone(String phone) {
        if (phone == null || phone.length() < 7) return phone;
        return phone.substring(0, 3) + "****" + phone.substring(7);
    }
    public void clearCode(String phone) { SMS_CODE_MAP.remove(phone); }
}

3 微信登录 — OAuth 2.0 流程与原理

微信登录采用标准 OAuth 2.0 授权码模式。理解这个流程是正确实现的前提。

完整时序图

  用户浏览器                          你的服务器                          微信开放平台
  ──────────                         ──────────                         ────────────

  ① 点击"微信登录"
  ─────────────────────────────────→ GET /api/wechat/authorize

                                     ② 生成 state + 拼装微信授权 URL
                                     ③ 302 重定向
  ←───────────────────────────────── Location: https://open.weixin.qq.com/connect/oauth2/authorize
                                       ?appid=wxAPPID
                                       &redirect_uri=https://你的域名/api/wechat/callback
                                       &response_type=code
                                       &scope=snsapi_login
                                       &state=RANDOM_TOKEN
                                       #wechat_redirect

  ④ 微信显示二维码 ──────────────────────────────────────────────────→
  ←─ 用户用微信扫码并确认授权 ─────────────────────────────────────────

  ⑤ 微信302回调你的服务器（携带 code）
  ←────────────────────────────────── GET /api/wechat/callback?code=AUTH_CODE&state=RANDOM_TOKEN

                                     ⑥ 验证 state（防CSRF）
                                     ⑦ 用 code 换 access_token
                                     ─────────────────────────────────→ GET /sns/oauth2/access_token
                                                                         ?appid=..&secret=..&code=..&grant_type=authorization_code

                                     ─────────────────────────────────→ GET /sns/userinfo

                                     ←─────────────────────────────────

                                     ⑪ 查库 → 登录/注册 → 生成自有JWT令牌
  ⑫ 302 跳转首页（附带 token）
  ←────────────────────────────────── Location: /index.html?token=JWT_TOKEN

4.2 微信服务类（新建） NEW 核心

封装与微信开放平台的全部交互：授权码换 token、拉取用户资料。使用 JDK 内置的 HttpURLConnection，不引入第三方 HTTP 库。

Javasrc/main/java/cn/puge/service/WechatService.java
package cn.puge.service;

import cn.puge.config.WechatConfig;
import cn.puge.exception.BusinessException;

import java.io.*;
import java.net.*;
import java.nio.charset.StandardCharsets;
import java.util.regex.Matcher;
import java.util.regex.Pattern;

/**
 * 微信登录服务
 * 负责与微信开放平台交互：获取 access_token、拉取用户信息
 * 全部使用 JDK 内置类，无需第三方依赖
 */
public class WechatService {

    // ================================================================
    // ① 用授权码（code）换取 access_token 和 openId
    // ================================================================
    public WechatTokenInfo getAccessToken(String code) {
        String url = WechatConfig.ACCESS_TOKEN_URL
            + "?appid=" + WechatConfig.APP_ID
            + "&secret=" + WechatConfig.APP_SECRET
            + "&code=" + urlEncode(code)
            + "&grant_type=authorization_code";

        String responseJson = httpGet(url);
        return parseTokenResponse(responseJson);
    }

    // ================================================================
    // ② 用 access_token 和 openId 获取微信用户信息
    // ================================================================
    public WechatUserInfo getUserInfo(String accessToken, String openId) {
        String url = WechatConfig.USER_INFO_URL
            + "?access_token=" + urlEncode(accessToken)
            + "&openid=" + urlEncode(openId)
            + "&lang=zh_CN";

        String responseJson = httpGet(url);
        return parseUserInfoResponse(responseJson);
    }

    // ========== HTTP GET（JDK 内置 HttpURLConnection）==========
    private String httpGet(String urlStr) {
        try {
            URL url = new URL(urlStr);
            HttpURLConnection conn = (HttpURLConnection) url.openConnection();
            conn.setRequestMethod("GET");
            conn.setConnectTimeout(5000);
            conn.setReadTimeout(5000);
            conn.setRequestProperty("Accept", "application/json");

            try (BufferedReader br = new BufferedReader(
                    new InputStreamReader(
                        conn.getResponseCode() == 200
                            ? conn.getInputStream()
                            : conn.getErrorStream(),
                        StandardCharsets.UTF_8))) {
                StringBuilder sb = new StringBuilder();
                String line;
                while ((line = br.readLine()) != null) { sb.append(line); }
                return sb.toString();
            }
        } catch (IOException e) {
            throw new BusinessException("WECHAT_HTTP_ERROR",
                "调用微信接口网络异常: " + e.getMessage());
        }
    }

    // ========== 解析 access_token 接口响应 ==========
    private WechatTokenInfo parseTokenResponse(String json) {
        if (json.contains("\"errcode\"")) {
            String errcode = extractJson(json, "errcode");
            String errmsg  = extractJson(json, "errmsg");
            throw new BusinessException("WECHAT_TOKEN_" + errcode,
                "换取access_token失败: " + errmsg);
        }

        WechatTokenInfo info = new WechatTokenInfo();
        info.accessToken  = extractJson(json, "access_token");
        info.openId       = extractJson(json, "openid");
        info.unionId      = extractJson(json, "unionid");     // 可能为 null
        info.refreshToken = extractJson(json, "refresh_token");

        String expiresStr = extractJson(json, "expires_in");
        if (expiresStr != null) info.expiresIn = Integer.parseInt(expiresStr);

        return info;
    }

    // ========== 解析 userinfo 接口响应 ==========
    private WechatUserInfo parseUserInfoResponse(String json) {
        if (json.contains("\"errcode\"")) {
            String errmsg = extractJson(json, "errmsg");
            throw new BusinessException("WECHAT_USERINFO_ERROR",
                "获取用户信息失败: " + errmsg);
        }

        WechatUserInfo info = new WechatUserInfo();
        info.openId    = extractJson(json, "openid");
        info.unionId   = extractJson(json, "unionid");
        info.nickname  = unescapeUnicode(extractJson(json, "nickname"));
        info.headImgUrl = extractJson(json, "headimgurl");
        info.sex        = extractJson(json, "sex");  // "1"=男 "2"=女 "0"=未知
        return info;
    }

    // ========== 简易 JSON 字段提取 ==========
    private String extractJson(String json, String key) {
        // 匹配 "key":"value"
        Pattern p = Pattern.compile(
            "\"" + key + "\"\\s*:\\s*\"([^\"]*)\"");
        Matcher m = p.matcher(json);
        if (m.find()) return m.group(1);

        // 匹配 "key":数字
        p = Pattern.compile("\"" + key + "\"\\s*:\\s*(-?\\d+)");
        m = p.matcher(json);
        if (m.find()) return m.group(1);

        return null;
    }

    // ========== Unicode 转义解码（微信昵称含 \uXXXX）==========
    private String unescapeUnicode(String str) {
        if (str == null) return null;
        StringBuilder sb = new StringBuilder();
        int i = 0;
        while (i < str.length()) {
            char c = str.charAt(i);
            if (c == '\\' && i + 5 < str.length() && str.charAt(i + 1) == 'u') {
                try {
                    int cp = Integer.parseInt(str.substring(i + 2, i + 6), 16);
                    sb.append((char) cp);
                    i += 6;
                    continue;
                } catch (NumberFormatException ignored) { }
            }
            sb.append(c);
            i++;
        }
        return sb.toString();
    }

    private String urlEncode(String value) {
        try { return URLEncoder.encode(value, "UTF-8"); }
        catch (UnsupportedEncodingException e) { return value; }
    }

    // ========== 内部数据类 ==========
    public static class WechatTokenInfo {
        public String accessToken;
        public String openId;
        public String unionId;
        public String refreshToken;
        public int    expiresIn;
    }

    public static class WechatUserInfo {
        public String openId;
        public String unionId;
        public String nickname;
        public String headImgUrl;  // 微信头像 URL（注意：不带 https 前缀）
        public String sex;         // "1"男 "2"女 "0"未知
    }
}

private LoginResponse handleWechatLogin(LoginRequest request) {
    String wechatOpenId;
    String nickname   = null;
    String avatarUrl  = null;
    String unionId    = null;

    // 路径A：已有 openId（历史授权用户，跨 session 恢复登录态）
    if (request.getWechatOpenId() != null
            && !request.getWechatOpenId().trim().isEmpty()) {
        wechatOpenId = request.getWechatOpenId();
    }
    // 路径B：标准 OAuth 流程 — 用授权码（code）换取
    else {
        // 步骤 ➊：code → access_token + openId
        WechatService.WechatTokenInfo tokenInfo =
            wechatService.getAccessToken(request.getWechatCode());

        wechatOpenId = tokenInfo.openId;
        unionId      = tokenInfo.unionId;

        // 步骤 ➋：access_token + openId → 用户昵称 / 头像
        WechatService.WechatUserInfo userInfo =
            wechatService.getUserInfo(tokenInfo.accessToken, wechatOpenId);

        nickname  = userInfo.nickname;
        avatarUrl = userInfo.headImgUrl;
        if (avatarUrl != null && !avatarUrl.startsWith("http")) {
            avatarUrl = "https:" + avatarUrl;  // 微信返回的头像不带协议前缀
        }
    }

    // 步骤 ➌：查库 — 根据 openId 查询/注册用户
    PugeUser user = getUserByWechatOpenId(wechatOpenId);
    if (user == null) {
        // 新用户：自动注册
        user = registerWechatUser(wechatOpenId, unionId, nickname, avatarUrl);
    } else {
        // 老用户：更新可能变化的资料
        if (nickname  != null) user.setNickname(nickname);
        if (avatarUrl != null) user.setAvatarUrl(avatarUrl);
        if (unionId   != null) user.setUnionId(unionId);
    }

    // 步骤 ➍：发放自有系统令牌
    return generateLoginResponse(user);
}

import java.util.UUID;
import cn.puge.config.WechatConfig;

// ================================================================
// 新增方法 ①：构造微信授权 URL → 302 跳转
// ================================================================
private void handleWechatAuthorize(com.sun.net.httpserver.HttpExchange exchange)
        throws IOException {

    // 生成随机 state 令牌（防 CSRF 攻击）
    // 生产环境中应存入 Redis/Session，回调时比对
    String state = UUID.randomUUID().toString().replace("-", "");

    // 拼装微信 OAuth 授权 URL
    String authUrl = WechatConfig.AUTHORIZE_URL
        + "?appid=" + WechatConfig.APP_ID
        + "&redirect_uri=" + URLEncoder.encode(WechatConfig.REDIRECT_URI, "UTF-8")
        + "&response_type=code"
        + "&scope=snsapi_login"      // 网站应用必须用 snsapi_login
        + "&state=" + state
        + "#wechat_redirect";            // 必须！不带的后果是微信不会回调

    // 302 重定向到微信
    exchange.getResponseHeaders().set("Location", authUrl);
    exchange.sendResponseHeaders(302, -1);
    exchange.close();
}

// ================================================================
// 新增方法 ②：处理微信回调 → 用 code 登录 → 302 回到前端
// ================================================================
private void handleWechatCallback(com.sun.net.httpserver.HttpExchange exchange)
        throws IOException {

    String query = exchange.getRequestURI().getQuery();
    Map<String, String> params = parseQueryString(query);
    String code  = params.get("code");
    String state = params.get("state");

    // TODO: 生产环境验证 state 与之前生成的随机令牌是否一致（防 CSRF）

    if (code == null || code.isEmpty()) {
        // 用户拒绝了授权，回到登录页
        exchange.getResponseHeaders().set("Location",
            "/login.html?error=user_denied");
        exchange.sendResponseHeaders(302, -1);
        exchange.close();
        return;
    }

    try {
        // 构造登录请求
        LoginRequest loginReq = new LoginRequest();
        loginReq.setLoginType(LoginRequest.LoginType.WECHAT.getCode());
        loginReq.setWechatCode(code);
        loginReq.setClientIp(
            exchange.getRemoteAddress().getAddress().getHostAddress());

        // 执行业务登录
        LoginResponse response = loginService.login(loginReq);

        if (response.getSuccess()) {
            // 登录成功 → 302 到首页，通过 URL 参数传递 token
            // 生产环境建议改为 Set-Cookie（HttpOnly + Secure）
            String redirect = "/index.html?token="
                + URLEncoder.encode(response.getAccessToken(), "UTF-8")
                + "&nickname="
                + URLEncoder.encode(response.getUserInfo().getNickname(), "UTF-8");
            exchange.getResponseHeaders().set("Location", redirect);
        } else {
            // 登录失败 → 回到登录页并展示错误信息
            exchange.getResponseHeaders().set("Location",
                "/login.html?error="
                    + URLEncoder.encode(response.getMessage(), "UTF-8"));
        }
    } catch (Exception e) {
        exchange.getResponseHeaders().set("Location",
            "/login.html?error=wechat_login_failed");
    }

    exchange.sendResponseHeaders(302, -1);
    exchange.close();
}

// ================================================================
// 新增辅助方法：解析 URL 查询参数
// ================================================================
private Map<String, String> parseQueryString(String query) {
    Map<String, String> params = new HashMap<>();
    if (query == null || query.isEmpty()) return params;

    String[] pairs = query.split("&");
    for (String pair : pairs) {
        int idx = pair.indexOf("=");
        if (idx > 0) {
            String key   = URLDecoder.decode(pair.substring(0, idx), StandardCharsets.UTF_8);
            String value = URLDecoder.decode(pair.substring(idx + 1), StandardCharsets.UTF_8);
            params.put(key, value);
        }
    }
    return params;
}

维度短信验证码登录微信登录
实现难度⭐⭐ 简单 — 核心只改 1 个方法⭐⭐⭐⭐ 中等 — 需完整 OAuth 流程

费用≈0.045 元/条免费
用户体验需等待短信（5-10秒）扫码即登（秒级）

获取信息量仅手机号OpenId + UnionId + 昵称 + 头像

推荐实施路线

  • 第一阶段：先上线短信验证码登录（改动最小、最可控、覆盖面最广）
  • 第二阶段：再接入微信登录（等企业资质和域名备案就绪后）
  • 第三阶段：打通用户体系（通过手机号或 unionId 将短信用户和微信用户关联到同一账号）
